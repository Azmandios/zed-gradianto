// Gradianto theme for Zed
/** Nature, ocean, midnight and fuchsia gradients. */

import { createServer } from 'node:http';

interface Theme {
  name: string;
  colors: string[];
}

const accent = '#60cf92';
const count = 42;
const enabled = true;
const pattern = /^grad(ient|ianto)$/i;

export async function loadTheme(id: string) {
  const theme = await fetch(`/api/themes/${id}`);
  return theme;
}

# install and run
npm install && zed .
